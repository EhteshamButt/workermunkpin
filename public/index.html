📖 Quick Start

Example: Generate QR Code

fetch('https://workerlocationpcikerweb.vercel.app/worker', {
  method: 'POST',
  headers: { 'Content-Type': 'application/json' },
  body: JSON.stringify({
    task: 'generateQRCode',
    payload: { text: 'https://munkpin.com', size: 200 }
  })
})
  .then(r => r.json())
  .then(console.log);
